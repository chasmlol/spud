# Record schema: worlds, cells, placed references, landscape, navigation, sky and lighting.
# The language is described in docs/formats/forms.md ("Schema"); shared types are in common.txt.
#
# Sources: xEdit's TES5/SSE definitions (Core/wbDefinitionsTES5.pas, wbDefinitionsCommon.pas), UESP's
# "Skyrim Mod:Mod File Format" pages, and the user's load order as checked by `skytool records check`.

# ---- Types -------------------------------------------------------------------------------------------------------

# Every record type a reference FormID may name (xEdit's reference list without its PLYR pseudo-record: the
# player reference 0x14 is created by the engine and has no record).
type WldRef = formid(ACHR,REFR,PGRE,PHZD,PMIS,PARW,PBAR,PBEA,PCON,PFLA)

# xEdit wbByteColors: red, green, blue and an unused byte.
type WldRgb = { r:u8 g:u8 b:u8 _:u8 }

# xEdit wbFloatRGBA.
type WldColorF4 = { r:f32 g:f32 b:f32 a:f32 }

# Directional ambient colours (xEdit wbAmbientColors): six axis colours, then specular and fresnel power.
# xEdit has the last two from form version 30, but the data has 24-byte DALCs at versions 30 and 31 (WTHR
# 0002E7AB, LGTM 0007545E): they are simply absent from shorter fields. Inside XCLL and LGTM DATA, which can
# end after the axis colours (CELL 00000025, LGTM 0007BA87), they are top-level members so the struct may end
# there.
type WldAmbientDirectional = { x_plus:WldRgb x_minus:WldRgb y_plus:WldRgb y_minus:WldRgb z_plus:WldRgb
                               z_minus:WldRgb }
type WldAmbientColors = { directional:WldAmbientDirectional specular:WldRgb fresnel_power:f32 }

# A colour for each of the four times of day (xEdit wbWeatherTimeOfDay, TES5).
type WldTimeOfDay = { sunrise:WldRgb day:WldRgb sunset:WldRgb night:WldRgb }

type WldPosRot = { position:Vec3 rotation:Vec3 }

# Ragdoll bone (XRGD): bone id, padding, position and rotation.
type WldRagdollBone = { bone_id:u8 _:bytes(3) position:Vec3 rotation:Vec3 }

# Enable parent (XESP): the parent, flags (0x01 opposite of parent, 0x02 pop in), padding.
type WldEnableParent = { reference:WldRef flags:u8 _:bytes(3) }

# Linked reference (XLKR): keyword (0: none), reference. Records of form version 20-25 store only the
# reference (4 bytes, e.g. REFR 0003EABA; xEdit: optional from member 1).
type WldLinkedRef = union(size) { 4: { reference:WldRef } *: { keyword:formid(KYWD) reference:WldRef } }

type WldActivateParent = { reference:WldRef delay:f32 }

# Reflected / refracted by (XPWR): the water reference, then (xEdit, optional from member 1) flags 0x01
# reflection, 0x02 refraction. Every XPWR in the load order (1402) is 4 bytes, the reference alone.
type WldWaterReflection = union(size) { 4: { reference:formid(REFR) } *: { reference:formid(REFR) type:u32 } }

# Patrol topic (PDTO): type 0 names a topic, type 1 a dialogue subtype code.
type WldTopicData = { type:u32 data:union(.type) { 0: formid(DIAL) 1: u32 } }

# Water current (XWCU element): velocity, then a float xEdit leaves unnamed.
type WldWaterCurrent = { velocity:Vec3 unknown:f32 }

# xEdit wbSizePosRot (XOCP occlusion plane, XPTL room portal): size, position, rotation quaternion.
type WldSizePosRot = { width:f32 height:f32 position:Vec3 rotation:{ q1:f32 q2:f32 q3:f32 q4:f32 } }

# Time interpolator key (IMAD): time, value.
type WldTimeKey = { time:f32 value:f32 }
# Colour interpolator key (IMAD): time, RGBA.
type WldColorKey = { time:f32 value:WldColorF4 }
# IMAD DNAM: number of keys in a field's mult and add interpolator arrays.
type WldMultAddCount = { mult_count:u32 add_count:u32 }

# Navmesh pathing cell (NVNM, NVMI): CRC of "PathingCell", the worldspace, then the grid cell of an exterior
# navmesh or the interior cell (xEdit wbNVNMParentDecider / wbNAVIParentDecider: by whether the world is
# none).
type WldPathingCell = { crc:u32 world:formid(WRLD)
                        cell:union(.world) { 0: formid(CELL) *: { grid_y:i16 grid_x:i16 } } }

# ---- Field sets --------------------------------------------------------------------------------------------------

fields WldOwnership
  XOWN owner formid(FACT,NPC_)
  XRNK faction_rank i32
end

fields WldRagdoll
  XRGD ragdoll_bones WldRagdollBone[]
  XRGB biped_rotation Vec3
end

fields WldActivateParents
  XAPD parent_activate_only u8
  XAPR activate_parent WldActivateParent
end

# Placed projectiles, hazards, missiles, beams, ... (xEdit TwbGameDefTES5.ReferenceRecord).
fields WldPlacedProjectile
  EDID editor_id zstring
  VMAD scripts Scripts
  NAME base formid(PROJ,HAZD)
  XEZN encounter_zone formid(ECZN)
  use WldOwnership
  use WldRagdoll
  XHTW head_tracking_weight f32
  XFVC favor_cost f32
  XPWR reflected_by WldWaterReflection
  XLKR linked_reference WldLinkedRef
  use WldActivateParents
  XESP enable_parent WldEnableParent
  XEMI emittance formid(LIGH,REGN)
  XMBR multibound_reference formid(REFR)
  XIS2 ignored_by_sandbox empty
  XLRT location_ref_types formid(LCRT)[]
  XLRL location_reference formid(LCRT,LCTN)
  XLOD distant_lod f32[3]
  XSCL scale f32
  DATA position WldPosRot
end

# ---- Records -----------------------------------------------------------------------------------------------------

# xEdit wbDefinitionsTES5 WRLD (wbDefinitionsCommon wbWorld* helpers).
record WRLD "Worldspace"
  EDID editor_id zstring
  # Large references by cell: the cell's grid, then each reference with its own cell grid.
  RNAM large_references { grid_y:i16 grid_x:i16
                          references:{ reference:formid(REFR) grid_y:i16 grid_x:i16 }[u32] }
  # Cell bounds (min, max), then 4 corner heights per cell, row by row: (max - min + 1) cells each way, which
  # the data matches in every worldspace that has one.
  MHDT max_height { min_x:i16 min_y:i16 max_x:i16 max_y:i16
                    cell_heights:{ bottom_left:u8 bottom_right:u8 top_left:u8 top_right:u8 }[] }
  FULL name lstring
  WCTR fixed_center { x:i16 y:i16 }
  LTMP interior_lighting formid(LGTM)
  XEZN encounter_zone formid(ECZN)
  XLCN location formid(LCTN)
  WNAM parent_world formid(WRLD)
  PNAM parent_flags u16
  CNAM climate formid(CLMT)
  NAM2 water formid(WATR)
  NAM3 lod_water formid(WATR)
  NAM4 lod_water_height f32
  DNAM land_data { default_land_height:f32 default_water_height:f32 }
  ICON map_image zstring
  use Model
  MNAM map_data { usable_dimensions:{ x:i32 y:i32 } nw_cell:{ x:i16 y:i16 } se_cell:{ x:i16 y:i16 }
                  camera:{ min_height:f32 max_height:f32 initial_pitch:f32 } }
  ONAM map_offset { scale:f32 cell_x_offset:f32 cell_y_offset:f32 cell_z_offset:f32 }
  NAMA distant_lod_multiplier f32
  DATA flags u8
  NAM0 bounds_min { x:f32 y:f32 }
  NAM9 bounds_max { x:f32 y:f32 }
  ZNAM music formid(MUSC)
  NNAM canopy_shadow zstring
  XNAM water_noise_texture zstring
  XWEM water_environment_map zstring
  TNAM hd_lod_diffuse_texture zstring
  UNAM hd_lod_normal_texture zstring
  # Offsets of the cells' records in the plugin, one per cell of the NAM0..NAM9 bounds (4096-unit cells), row
  # by row (as the data has in every worldspace); the game never reads them.
  OFST offsets u32[]
end

# xEdit wbDefinitionsTES5 CELL.
record CELL "Cell"
  EDID editor_id zstring
  FULL name lstring
  # One byte in some form version 18 records (e.g. 00023779; xEdit: "Flags can be itU8").
  DATA flags union(size) { 1: u8 *: u16 }
  XCLC grid { x:i32 y:i32 land_flags:u8 _:bytes(3) }
  # Ends after the ambient axis colours in CELL 00000025 (form version 30, 64 bytes).
  XCLL lighting { ambient:WldRgb directional:WldRgb fog_near_color:WldRgb fog_near:f32 fog_far:f32
                  directional_rotation_xy:i32 directional_rotation_z:i32 directional_fade:f32
                  fog_clip_distance:f32 fog_power:f32 ambient_colors:WldAmbientDirectional
                  ambient_specular:WldRgb ambient_fresnel_power:f32 fog_far_color:WldRgb fog_max:f32
                  light_fade_begin:f32 light_fade_end:f32 inherits:u32 }
  TVDT terrain_visibility bytes
  MHDT max_height { offset:f32 max_heights:u8[32][32] }
  LTMP lighting_template formid(LGTM)
  # Leftover lighting flags (xEdit: "now in XCLC"); never read.
  LNAM unknown u32
  XCLW water_height f32
  XNAM water_noise_texture zstring
  XCLR regions formid(REGN)[]
  XLCN location formid(LCTN)
  XWCN water_velocity_count u32
  XWCS water_velocity_count_old u32
  XWCU water_velocities WldWaterCurrent[]
  XCWT water formid(WATR)
  use WldOwnership
  XILL lock_list formid(FLST,NPC_)
  XWEM water_environment_map zstring
  XCCM sky_region formid(REGN)
  XCAS acoustic_space formid(ASPC)
  XEZN encounter_zone formid(ECZN)
  XCMO music formid(MUSC)
  XCIM image_space formid(IMGS)
end

# xEdit wbDefinitionsTES5 REFR. INAM is an image space in room bound data (after XRMR) and an idle in patrol
# data (after XPRD).
record REFR "Placed object"
  EDID editor_id zstring
  VMAD scripts Scripts
  # xEdit's list. One reference breaks it: 000A9D18 (Skyrim.esm) names the spell 0004D3F8.
  NAME base formid(ACTI,ADDN,ALCH,AMMO,APPA,ARMO,ARTO,ASPC,BOOK,CONT,DOOR,FLOR,FURN,IDLM,INGR,KEYM,LIGH,MISC,
                   MSTT,SCOL,SCRL,SLGM,SOUN,STAT,TACT,TREE,TXST,WEAP)
  XMBO bound_half_extents Vec3
  XPRM primitive { bounds:Vec3 color:WldColorF4 type:u32 }
  XORD linked_occlusion_refs formid(REFR)[4]
  XOCP occlusion_plane WldSizePosRot
  XPOD portals { origin:formid(REFR) destination:formid(REFR) }[]
  XPTL room_portal WldSizePosRot
  XRMR room { linked_rooms_count:u8 flags:u8 unknown:bytes(2) } -> $ctx=1
  LNAM lighting_template formid(LGTM)
  INAM image_space formid(IMGS) when $ctx=1
  XLRM linked_room formid(REFR)
  XMBP multibound_primitive empty
  use WldRagdoll
  XRDS radius f32
  XPWR reflected_by WldWaterReflection
  XLTW lit_water formid(REFR)
  XEMI emittance formid(LIGH,REGN)
  # 16 bytes in form version 35 and 37 records, 20 from 43 (xEdit: optional from member 4).
  XLIG light { fov:f32 fade:f32 end_distance_cap:f32 shadow_depth_bias:f32 unknown:bytes(4) }
  XALP alpha { cutoff:u8 base:u8 }
  XTEL teleport { door:formid(REFR) position:Vec3 rotation:Vec3 no_alarm:u32 }
  XTNM teleport_message formid(MESG)
  XMBR multibound_reference formid(REFR)
  XWCN water_velocity_count u32
  XWCS water_velocity_count_old u32
  XWCU water_velocities WldWaterCurrent[]
  XCVL water_linear_velocity Vec3
  XCVR water_rotational_velocity Vec3
  XCZR water_current_reference WldRef
  XCZC water_current_cell formid(CELL)
  XCZA water_current_unknown u32
  XSCL scale f32
  XSPC spawn_container formid(REFR)
  use WldActivateParents
  XLIB leveled_item_base formid(LVLI)
  XLCM level_modifier i32
  XLCN persist_location formid(LCTN)
  # Index into the collision layer (COLL) records, not a FormID.
  XTRI collision_layer u32
  XLOC lock { level:u8 _:bytes(3) key:formid(KEYM) flags:u8 _:bytes(3) _:bytes(8) }
  XEZN encounter_zone formid(ECZN)
  XNDP navmesh_door_link { navmesh:formid(NAVM) triangle:i16 _:bytes(2) }
  XLRT location_ref_types formid(LCRT)[]
  XIS2 ignored_by_sandbox empty
  use WldOwnership
  XCNT item_count i32
  XCHG charge f32
  XLRL location_reference formid(LCRT,LCTN)
  XESP enable_parent WldEnableParent
  XLKR linked_reference WldLinkedRef
  XPRD patrol_idle_time f32 -> $ctx=2
  XPPA patrol_script_marker empty
  INAM patrol_idle formid(IDLE) when $ctx=2
  # Leftovers of earlier CK versions (xEdit TES5: unused): a script header in xEdit's Fallout 3 SCHR layout
  # (data: 20 bytes, enabled = 1) and an empty script source.
  SCHR unused_script_header { _:bytes(4) ref_count:u32 compiled_size:u32 variable_count:u32 type:u16
                              enabled:u16 }
  SCTX unused_script_source zstring
  PDTO patrol_topic WldTopicData
  XACT action_flags u32
  XHTW head_tracking_weight f32
  XFVC favor_cost f32
  ONAM open_by_default empty
  XMRK map_marker empty
  FNAM map_flags u8
  FULL map_name lstring
  TNAM map_marker_type { type:u8 _:u8 }
  XATR attach_ref formid(REFR,PGRE,PHZD,PMIS,PARW,PBAR,PBEA,PCON,PFLA)
  XLOD distant_lod f32[3]
  DATA position WldPosRot
end

# xEdit wbDefinitionsTES5 LAND (wbLand* helpers). 33 x 33 vertices per cell. ATXT is followed by its VTXT.
record LAND "Landscape"
  DATA flags u32
  # Normals are signed: the data has e.g. (0xFD, 0x01, 0x7E) for a near-vertical normal, unit length only as
  # signed bytes (xEdit types the components u8).
  VNML normals { x:i8 y:i8 z:i8 }[33][33]
  VHGT heights { offset:f32 deltas:i8[33][33] _:bytes(3) }
  VCLR colors { r:u8 g:u8 b:u8 }[33][33]
  BTXT base_layer { texture:formid(LTEX) quadrant:u8 _:u8 layer:i16 }
  ATXT alpha_layer { texture:formid(LTEX) quadrant:u8 _:u8 layer:i16 }
  VTXT alpha_layer_data { position:u16 _:bytes(2) opacity:f32 }[]
  MPCD unknown bytes
end

# xEdit wbDefinitionsTES5 NAVM / wbNVNM (navmesh version 12).
record NAVM "Navmesh"
  EDID editor_id zstring
  NVNM geometry { version:u32 pathing_cell:WldPathingCell
                  vertices:Vec3[u32]
                  triangles:{ vertices:u16[3] edges:i16[3] flags:u16 cover_flags:u16 }[u32]
                  edge_links:{ type:u32 navmesh:formid(NAVM) triangle:i16 }[u32]
                  door_links:{ triangle:i16 crc:u32 door:formid(REFR) }[u32]
                  cover_triangles:i16[u32]
                  # Grid of divisor x divisor cells, each the triangles it overlaps, to the end of the
                  # field (exactly divisor^2 cells in all 19393 navmeshes; every vertex and triangle index
                  # in range).
                  grid:{ divisor:u32 cell_size:{ x:f32 y:f32 } bounds_min:Vec3 bounds_max:Vec3
                         cells:i16[u32][] } }
  # xEdit leaves the target open; the data has statics only, apart from stale entries in navmeshes of the test
  # cell 00000025 (REFRs, the cell itself and non-FormIDs such as 0x3F in NAVM 0001CD4A).
  ONAM base_objects formid(STAT)[]
  PNAM preferred_connectors u16[]
  NNAM non_connectors u16[]
end

# xEdit wbDefinitionsTES5 NAVI.
record NAVI "Navmesh info map"
  EDID editor_id zstring
  NVER version u32
  NVMI navmesh_info { navmesh:formid(NAVM) flags:u32 approx_location:Vec3 preferred_percent:f32
                      edge_links:formid(NAVM)[u32] preferred_edge_links:formid(NAVM)[u32]
                      door_links:{ crc:u32 door:formid(REFR) }[u32]
                      has_island_data:u8
                      island:union(.has_island_data) {
                        0: empty
                        *: { bounds_min:Vec3 bounds_max:Vec3 triangles:{ v0:u16 v1:u16 v2:u16 }[u32]
                             vertices:Vec3[u32] } }
                      pathing_cell:WldPathingCell }
  NVPP precomputed_pathing { paths:formid(NAVM)[u32][u32]
                             road_markers:{ navmesh:formid(NAVM) index:u32 }[u32] }
  NVSI deleted_navmeshes formid(NAVM)[]
end

# xEdit wbDefinitionsTES5 WATR. DNAM is 228 bytes before SSE's flowmap scale (xEdit: optional from member 36;
# data: 228 at form version 39, 232 at 44).
record WATR "Water"
  EDID editor_id zstring
  FULL name lstring
  NNAM old_noise_texture zstring
  ANAM opacity u8
  FNAM flags u8
  MNAM material_id u8
  TNAM material formid(MATT)
  SNAM open_sound formid(SNDR)
  XNAM spell formid(SPEL)
  INAM image_space formid(IMGS)
  DATA damage_per_second u16
  DNAM data { wind_velocity:f32 wind_direction:f32 wave_amplitude:f32 wave_frequency:f32
              sun_specular_power:f32 reflectivity_amount:f32 fresnel_amount:f32 unknown:f32
              fog_above_near:f32 fog_above_far:f32 shallow_color:WldRgb deep_color:WldRgb
              reflection_color:WldRgb texture_blend:u8 _:bytes(3)
              rain_simulator:f32[4]
              displacement:{ starting_size:f32 force:f32 velocity:f32 falloff:f32 dampner:f32 }
              unknown2:f32 noise_falloff:f32 noise_wind_direction:f32[3] noise_wind_speed:f32[3]
              unknown3:f32 unknown4:f32 fog_above_amount:f32 unknown5:f32
              fog_under:{ amount:f32 near:f32 far:f32 }
              refraction_magnitude:f32 specular_power:f32 unknown6:f32 specular_radius:f32
              specular_brightness:f32 noise_uv_scale:f32[3] noise_amplitude_scale:f32[3]
              reflection_magnitude:f32 sun_sparkle_magnitude:f32 sun_specular_magnitude:f32
              depth:{ reflections:f32 refraction:f32 normals:f32 specular_lighting:f32 }
              sun_sparkle_power:f32 noise_flowmap_scale:f32 }
  GNAM related_waters { daytime:formid(WATR) nighttime:formid(WATR) underwater:formid(WATR) }
  NAM0 linear_velocity Vec3
  NAM1 angular_velocity Vec3
  NAM2 noise_texture1 zstring
  NAM3 noise_texture2 zstring
  NAM4 noise_texture3 zstring
  NAM5 flow_normals_noise_texture zstring
end
